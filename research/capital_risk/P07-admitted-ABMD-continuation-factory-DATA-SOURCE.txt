FROZEN_SHA='4f3db7f1cefffbad4c6eb48c0efe82678282c96c2a7b97c039db2307b9d86baa'
SCHEMA='research.p07-abmd-admitted-continuation-qualification/1'

def encoded(value):return (json.dumps(value,sort_keys=True,indent=2,allow_nan=False)+'\n').encode()
def proof_bytes(rows):
    from system_campaign_financial_branches import encoded as original_encoded
    return b''.join(original_encoded(row)+b'\n' for row in rows)

def factory_tree(raw):
    need(sha(raw)==FROZEN_SHA,'P07_ABMD_CONTINUATION_EXACT_FROZEN_CERTIFICATE_SOURCE')
    fn=models.legacy.original.function(raw,'extend_joint')
    cls,=[n for n in fn.body if isinstance(n,ast.ClassDef) and n.name=='ABMD']
    init,=[n for n in cls.body if isinstance(n,ast.FunctionDef) and n.name=='__init__']
    guard,=[n.value for n in init.body if isinstance(n,ast.Expr) and isinstance(n.value,ast.Call)
        and any(isinstance(v,ast.Constant) and v.value=='P07_ABMD_ORIGINAL_COMMITTED_SEED_REQUIRED' for v in n.value.args)]
    need(ast.unparse(guard.args[0])=='self.seed and self.seed[-1][\'session\'] < NATIVE',
         'P07_ABMD_CONTINUATION_ONE_EXACT_ORIGINAL_PRE_EVENT_SEED_GUARD')
    guard.args[0]=ast.parse('qualified_retained_seed(self.seed,event,source,codec)',mode='eval').body
    return ast.fix_missing_locations(fn)

def qualify_metadata(current,spec,catalog,origin_value,rows,committed,resolutions,allowed,archive_raw,*,source_plan):
    """Metadata only. Production calls follow actual admission/origin/seed guards."""
    p=source_plan;day=catalog['event']['effective_session']
    need(spec['arm']=='P07' and spec['root_admission_sha256']!=lifecycle.INITIAL_ROOT
        and lifecycle.valid_retained(spec) and len(current['repairs'])==24
        and current['repairs']==spec['repairs'] and current['repairs'][:23]==p['parent_descriptors']
        and current['base']==spec['base'] and current['sha256']==spec['outer_sha256']
        and current['completed']==spec['measured_closes'] and current['source_cursor']==spec['selected_session']
        and current['source_cursor']>=day and current['run']['input_sha256']==spec['input_sha256']
        ==digest(dict(base=current['base'],repairs=current['repairs']))
        and current['run']['checkpoint_sha256']==spec['inner_sha256'],
        'P07_ABMD_CONTINUATION_ACTUAL_ADMITTED_FULL24_CURRENT_IDENTITY')
    need(origin_value['schema']==origin.SCHEMA and origin_value['passed'] is True
        and origin_value['initial_root_admission_sha256']==lifecycle.INITIAL_ROOT
        and origin_value['catalog_sha256']==lifecycle.CATALOG_SHA
        and origin_value['descriptor']==current['repairs'][23]==spec['retained_abmd']['descriptor']
        and sha(encoded(origin_value))==spec['retained_abmd']['origin_sha256'],
        'P07_ABMD_CONTINUATION_EXACT_PRIOR_NEUTRAL_ORIGIN_BYTES')
    proof=spec['root_admission']['branch_uncertainty_proof']
    need(proof['passed'] is proof['all_comparisons_and_state_sets_verified']
        is proof['all_math_watch_financial_chains_verified'] is True
        and type(proof['rows']) is type(proof['retained_rows']) is int
        and proof['rows']>0 and proof['retained_rows']>=784
        and len(rows)==len(committed)==proof['rows']+proof['retained_rows']
        and rows[-1]['session']==current['source_cursor']
        and rows[-1]['next_state_sha256']==current['run']['state_sha256']==spec['state_sha256']
        and proof_bytes(rows)==proof_bytes(committed),
        'P07_ABMD_CONTINUATION_LITERAL_ACTUAL_ALL_CURRENT_COMMITTED_SEED')
    prefix=proof_bytes(rows[:784])
    need(dict(bytes=len(prefix),sha256=sha(prefix))==p['committed_certificate_prefix'],
        'P07_ABMD_CONTINUATION_ORIGINAL784_LITERAL_PREFIX')
    events=[row for row in rows if row['session']==day]
    need(len(events)==len(resolutions)==1,'P07_ABMD_CONTINUATION_EXACTLY_ONE_PRIOR_COMMITTED_ABMD_EVENT')
    row,=events;resolution,=resolutions;basis=row['bounded_event_basis']
    expected={digest(row):(day,catalog['event']['security_id'],basis['signal_consideration_interval'])} if basis['selected'] else {}
    need(allowed==expected and basis['event_contract_sha256']==digest(catalog['event'])
        and basis['source_event_sha256']==digest(catalog['market_source'])
        and resolution['actual']=='book/pre-event/'+day+'-'+resolution['outer_sha256']+'.json.gz'
        and resolution['state_sha256']==row['prior_state_sha256']
        and resolution['input_sha256']==row['input_sha256']==current['run']['input_sha256']
        and sha(archive_raw)==resolution['gzip_sha256'],
        'P07_ABMD_CONTINUATION_ONE_EXACT_PRIOR_EVENT_SOURCE_AND_PRODUCER_BYTES')
    raw=proof_bytes(rows)
    return dict(schema=SCHEMA,passed=True,root_admission_sha256=spec['root_admission_sha256'],
        current_outer_sha256=current['sha256'],full_input_descriptors=24,committed_proof_rows=len(rows),
        seed_bytes=len(raw),seed_sha256=sha(raw),origin_sha256=sha(encoded(origin_value)),
        original784_prefix=p['committed_certificate_prefix'],prior_event_resolution=resolution,
        second_event_authorized=False,prior_proof_rows_reexecuted=0,financial_authority=False)

def compile_factory(qualified_retained_seed):
    raw=Path(frozen.__file__).read_bytes();scope=dict(vars(frozen),qualified_retained_seed=qualified_retained_seed)
    exec(compile(ast.Module(body=[factory_tree(raw)],type_ignores=[]),
        '<whole-frozen-ABMD-factory-one-admitted-retained-seed-predicate>','exec'),scope)
    return scope['extend_joint']

def certificate_class(event,source,codec,native,*,mixed,inherited_root,retained_root,current_checkpoint,admitted_spec,catalog):
    from full_system_core import hosted
    hosted()
    from system_campaign_financial_partial import validate_root_spec
    from system_campaign_financial_branches import seed
    from system_campaign_bounded_sensor_event_review import permissions
    from system_campaign_abmd_bounded_catalog import evidence
    from system_campaign_uncertainty_math import Interval
    from research.pooled_vacancy_funding import backtest,journal
    root=Path(retained_root);current=current_checkpoint;spec=admitted_spec
    validate_root_spec(spec)
    selected=(root/spec['selected_checkpoint']).read_bytes()
    need(dict(bytes=len(selected),sha256=sha(selected))==spec['selected_file'],
         'P07_ABMD_CONTINUATION_ACTUAL_ADMITTED_SELECTED_GZIP_BEFORE_PARSE')
    need(journal.read_checkpoint(root/spec['selected_checkpoint'])==current,
         'P07_ABMD_CONTINUATION_ACTUAL_CURRENT_NOT_CALLER_SUBSTITUTE')
    need(catalog==lifecycle.catalog() and event==catalog['event'] and source==catalog['market_source'],
         'P07_ABMD_CONTINUATION_UNCHANGED_V2_TYPED_INTERPRETATION')
    lifecycle.check_retained_origin(root,current,catalog,spec)
    models.validate_full(root,current,catalog,evidence(catalog),input_hash=backtest.input_hash,
        root_metadata=lifecycle.initial_spec())
    committed=seed(root,current)
    allowed,resolutions=permissions(root,committed,catalog,through=current['source_cursor'],input_descriptors=current['repairs'])
    need(len(resolutions)==1,'P07_ABMD_CONTINUATION_ACTUAL_ORIGINAL_EVENT_ARCHIVE_REQUIRED')
    archive=(root/resolutions[0]['actual']).read_bytes();value=json.loads((root/origin.ORIGIN).read_bytes())
    # The original factory is qualified before any instance construction.
    initial=frozen.certificate_class(event,source,codec,native,mixed=mixed,inherited_root=inherited_root)
    def qualified_retained_seed(rows,actual_event,actual_source,actual_codec):
        need(actual_event==event and actual_source==source and actual_codec is codec,
             'P07_ABMD_CONTINUATION_EXACT_FACTORY_CODEC_EVENT_SOURCE')
        need((root/origin.ORIGIN).read_bytes()==encoded(value),
             'P07_ABMD_CONTINUATION_ORIGINAL_NEUTRAL_ORIGIN_FRESH_BYTES')
        actual_archive=(root/resolutions[0]['actual']).read_bytes()
        need(actual_archive==archive,'P07_ABMD_CONTINUATION_ACTUAL_PRIOR_EVENT_ARCHIVE_FRESH_BYTES')
        qualify_metadata(current,spec,catalog,value,rows,committed,resolutions,allowed,actual_archive,source_plan=models.plan())
        return True
    continued=compile_factory(qualified_retained_seed)(initial.__mro__[1],event,source,codec,native,
        witness=initial.witness,interval_class=Interval)
    need(continued.append is initial.append and continued.__mro__[1:] == initial.__mro__[1:]
        and continued.witness is initial.witness,'P07_ABMD_CONTINUATION_ALL_ORIGINAL_COOPERATIVE_BINDINGS')
    return continued
